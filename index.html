function guardarCita() { const f = document.getElementById('agenda-fecha').value; const h = document.getElementById('agenda-hora').value; const cli = cleanStr(document.getElementById('agenda-cli').value).toUpperCase(); const cel = cleanNum(document.getElementById('agenda-cel').value); const barb = document.getElementById('agenda-barbero').value; if (!f || !h || !cli || !barb || tempServicios.length === 0) return Swal.fire('Aviso', 'Fecha, hora, cliente, barbero y servicios obligatorios.', 'warning'); const b = db.barberias[bIdx]; const nTime = timeToMins(h); const apertMins = timeToMins(b.horaApertura || "09:00"); const cierrMins = timeToMins(b.horaCierre || "21:00"); if (nTime < apertMins || nTime > cierrMins) return Swal.fire('Cerrado', `El local atiende de ${b.horaApertura} a ${b.horaCierre}`, 'error'); // 🛡️ RADAR ANTI-CHOQUES INTELIGENTE let totalMinsCarrito = 0; tempServicios.forEach(s => { totalMinsCarrito += parseDuracionToMins(s.dur); }); if (totalMinsCarrito === 0) totalMinsCarrito = (b.intervaloCitas || 60); const miInicio = timeToMins(h); const miFin = miInicio + totalMinsCarrito; const coll = (b.citas || []).find(c => { if (c.barbero !== barb || c.fecha !== f || c.estado === 'CANCELADA') return false; let cInicio = timeToMins(c.hora); let cFin = cInicio + (c.duracion || b.intervaloCitas || 60); return (miInicio < cFin && miFin > cInicio); }); if (coll) return Swal.fire('¡Horario Ocupado! ⏰', `${barb} ya tiene una cita reservada a las ${formatTime12h(coll.hora)} que choca con este servicio.`, 'error'); if (!b.citas) b.citas = []; // 🚀 PREPARAMOS LA CITA let srvString = tempServicios.map(s=>s.nom || s).join(' + '); let idCitaNueva = Date.now().toString(); b.citas.push({ id: idCitaNueva, nombre: cli, celular: cel, barbero: barb, servicio: srvString, fecha: f, hora: h, estado: 'PENDIENTE', duracion: totalMinsCarrito }); // ⚡ PUENTE SQL: DISPARO A SUPABASE EN SEGUNDO PLANO if (isOnline && typeof supabaseClient !== 'undefined' && b.id_nube) { // 1. Inserción en la tabla SQL (appointments) supabaseClient.from('appointments').insert([{ store_id: b.id_nube, customer_name: cli, customer_phone: cel || null, staff_name: barb, service_name: srvString, appointment_date: f, appointment_time: h, status: 'PENDIENTE', duration: totalMinsCarrito, receipt_url: null }]).then(({error}) => { if(error) console.error("❌ Error guardando cita en SQL:", error); else console.log("☁️ Cita inyectada a la tabla SQL exitosamente."); }); // 2. 🔔 GATILLO DE NOTIFICACIÓN PUSH AL ADMIN console.log("Revisando token Push para enviar notificación..."); let tokenPushAdmin = b.pushSubscription; if (tokenPushAdmin) { console.log("✅ Token Push encontrado. Intentando invocar Edge Function..."); // 🚀 USAMOS EL INVOCADOR NATIVO DE SUPABASE PARA EVITAR BLOQUEOS CORS/401 supabaseClient.functions.invoke('enviar-push', { body: { subscription: tokenPushAdmin, title: '¡NUEVA CITA KIOSKO! ⚡', body: `📅 ${f} | ⏰ ${formatTime12h(h)}\n👤 ${cli}\n💈 ${barb}` } }).then(({ data, error }) => { if (error) console.error('❌ Error disparando Push (Supabase Invoke):', error); else console.log('🔔 Push disparado con éxito:', data); }).catch(err => { console.error('Fallo crítico al invocar función Push:', err); }); } else { console.warn("⚠️ No se disparó Push: El local (Kiosko) no tiene una antena (pushSubscription) registrada en su base de datos local. ¿Se sincronizó el Kiosko después de que el Admin activara las alertas?"); } } // 👇 GUARDADO LOCAL (A prueba de fallos y modo Offline) const idCli = cel || cli; if (!b.clientes) b.clientes = {}; if (!b.clientes[idCli]) b.clientes[idCli] = { nombre: cli, celular: cel, pts: 0, visitas: 0 }; saveDB(true); // 🧹 LIMPIEZA PROFUNDA DEL FORMULARIO document.getElementById('agenda-cli').value = ""; document.getElementById('agenda-cel').value = ""; document.getElementById('agenda-fecha').value = ""; document.getElementById('agenda-hora').value = ""; document.getElementById('agenda-barbero').value = ""; actualizarFotoBarbero('agenda'); tempServicios = []; renderServiciosTemp('agenda'); renderAgendaList(); renderAdmin(); actualizarBadgesCitas(); Swal.fire({ title: '¡Cita Guardada!', icon: 'success', background: '#0a0a0a', color: '#fff', timer: 1500 }); }
